#### 2. Margin Protection Integration

typescript
// OrderNexus consumes SKU OS margin intelligence
class OrderMarginGuard {
  async evaluateOrderWithProductContext(order: Order): Promise<MarginAssessment> {
    const productContext = await this.getProductHealthContext(order.lineItems);

    return {
      baseProfitability: await this.calculateBaseProfit(order),
      productRisks: await this.assessProductLevelRisks(order, productContext),
      recommendations: await this.generateMarginProtectionStrategies(order, productContext)
    };
  }
  
  private async assessProductLevelRisks(order: Order, context: ProductHealthContext) {
    const risks = [];

    for (const item of order.lineItems) {
      const productHealth = context.get(item.productId);
      
      // Use SKU OS intelligence for margin protection
      if (productHealth?.marginHealth === 'critical') {
        risks.push({
          productId: item.productId,
          type: 'CRITICAL_MARGIN_PRODUCT',
          risk: 'This product has critically low margins',
          action: 'Consider price increase or removal from promotions'
        });
      }
      
      if (productHealth?.stockoutRisk > 0.8) {
        risks.push({
          productId: item.productId, 
          type: 'HIGH_STOCKOUT_RISK',
          risk: 'This product may go out of stock soon',
          action: 'Limit promotions to preserve inventory'
        });
      }
    }
    
    return risks;
  }
}

// packages/demand-forecasting/src/order-enhanced-predictor.ts
export class OrderEnhancedDemandPredictor {
  async forecastWithOrderHistory(productId: number): Promise<EnhancedDemandForecast> {
    const [baseHealth, orderPatterns, seasonality, promotions] = await Promise.all([
      this.healthScorer.getHealthScore(productId),
      this.orderAnalyzer.getProductOrderPatterns(productId),
      this.seasonalityDetector.getSeasonalFactors(productId),
      this.promotionAnalyzer.getPromotionImpact(productId)
    ]);

    return {
      baseDemand: baseHealth.demand_velocity_30d,
      enhancedForecast: this.calculateEnhancedForecast(orderPatterns, seasonality),
      confidence: this.calculateForecastConfidence(orderPatterns.historyLength),
      reorderRecommendation: await this.generateIntelligentReorder(productId, baseHealth, orderPatterns),
      stockoutProbability: this.calculateTimeBasedStockoutRisk(baseHealth, orderPatterns)
    };
  }

  private calculateEnhancedForecast(patterns: OrderPatterns, seasonality: SeasonalFactors): number {
    // Simple weighted average for Phase 2
    const base = patterns.unit_sales_30d;
    const trend = this.calculateTrend(patterns.unit_sales_7d, patterns.unit_sales_30d);
    const seasonalAdjustment = seasonality.currentMultiplier;

    return Math.max(0, base * (1 + trend) * seasonalAdjustment);
  }
}
